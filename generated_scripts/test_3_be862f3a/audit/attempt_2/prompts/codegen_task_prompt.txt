async def crawl_detail_direct(context, url, flags, records, args):
    item, blocked = await extract_detail(context, url, "DIRECT_INPUT", args.mask_reviews, flags)
    if item:
        records.append(item)
    return blocked


def input_rows(path):
    df = pd.read_excel(path, header=None).fillna("")
    rows = []
    for _, row in df.iterrows():
        vals = [clean(x) for x in row.tolist()]
        vals += [""] * (3 - len(vals))
        if any(vals[:3]):
            rows.append({"keyword": vals[0], "listing_url": vals[1], "detail_url": vals[2]})
    return rows
