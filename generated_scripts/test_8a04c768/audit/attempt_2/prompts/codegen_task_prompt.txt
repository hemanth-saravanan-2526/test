def read_input(path):
    sheets = pd.read_excel(path, sheet_name=None)
    sheets = {k: v.fillna("") for k, v in sheets.items()}
    input_df = sheets.get("Extracted Data") or next(iter(sheets.values()), pd.DataFrame())
    allowed = {"keyword", "listing_url", "detail_url"}
    if not any(str(c).strip().lower() in allowed for c in input_df.columns):
        input_df = pd.read_excel(path, header=None).fillna("")
        input_df = input_df.rename(columns={input_df.columns[0]: "keyword"})
    input_df.columns = [str(c).strip().lower() for c in input_df.columns]
    for col in allowed:
        if col not in input_df.columns:
            input_df[col] = ""
    return input_df


def export_results(scraper, output):
    columns = [
        "product_id", "product_url", "source_listing_url", "Nº", "Data", "Nome", "Tipologia",

        "Entidade Exploradora", "NIF / NIPC"
    ]
    data = []
    for record in scraper.records:

        pd.DataFrame(flags, columns=["Type", "URL", "Message"]).to_excel(writer, sheet_name="Flags", index=False)
